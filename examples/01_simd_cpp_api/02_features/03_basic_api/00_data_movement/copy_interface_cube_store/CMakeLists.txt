cmake_minimum_required(VERSION 3.16)
set(CMAKE_ASC_RUN_MODE "npu" CACHE STRING "Run mode")
set(CMAKE_ASC_ARCHITECTURES "dav-3510" CACHE STRING "NPU architecture")
set(COPY_ROUTE "0" CACHE STRING "0:L0C2GM 1:L0C2UB 2:L0C2L1")
set(SCENARIO_NUM "0" CACHE STRING "0:atom+coord 1:no-atom whole 2:no-atom+coord")
set(QUANT_MODE "0" CACHE STRING "0:none 1:scalar 2:tensor; only for L0C2GM")
if(NOT COPY_ROUTE STREQUAL "0" AND NOT QUANT_MODE STREQUAL "0")
  message(FATAL_ERROR "QUANT_MODE 1/2 is only valid for COPY_ROUTE=0")
endif()
find_package(ASC REQUIRED)
project(copy_interface_cube_store LANGUAGES ASC CXX)
add_executable(demo copy_interface_cube_store.asc)
set(ASCEND_HOME $ENV{ASCEND_HOME_PATH})
include_directories("${ASCEND_HOME}/asc/")
target_compile_definitions(demo PRIVATE COPY_ROUTE=${COPY_ROUTE} SCENARIO_NUM=${SCENARIO_NUM} QUANT_MODE=${QUANT_MODE})
target_compile_options(demo PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>)
