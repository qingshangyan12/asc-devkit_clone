cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(archive_cube_regression LANGUAGES ASC CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}"
    "${REPO}/impl/adv_api/cmake/scripts/gen_kernel_tiling_data_def.py"
    "${REPO}/include/adv_api" "${GENERATED}/kernel_tiling/kernel_tiling.h"
    RESULT_VARIABLE GEN_RESULT)
if(NOT GEN_RESULT EQUAL 0)
    message(FATAL_ERROR "Official tiling header generator failed")
endif()
if(CUBE_SYNC_TEST)
    add_executable(sync_test sync.asc)
    target_compile_options(sync_test PRIVATE --npu-arch=dav-3510
        -iquote${REPO}/include -iquote${REPO} -iquote${REPO}/include/basic_api -iquote${REPO}/include/utils)
    target_include_directories(sync_test PRIVATE "${GENERATED}")
    return()
endif()
if(NOT SAMPLE_DIR)
    set(CUBE_POINTER_MODE 0 CACHE STRING "0 Tensor, 1 pointer, 2 mixed")
    set(CUBE_SIDE 16 CACHE STRING "Aligned square matrix dimension")
    option(CUBE_BFLOAT16 "Use BF16 input" OFF)
    add_executable(cube_pointer_validation runner.asc)
    target_compile_definitions(cube_pointer_validation PRIVATE CUBE_POINTER_MODE=${CUBE_POINTER_MODE} CUBE_SIDE=${CUBE_SIDE})
    if(CUBE_BFLOAT16)
        target_compile_definitions(cube_pointer_validation PRIVATE CUBE_BFLOAT16=1)
    endif()
    target_compile_options(cube_pointer_validation PRIVATE --npu-arch=dav-3510
        -iquote${REPO}/include -iquote${REPO} -iquote${REPO}/include/basic_api -iquote${REPO}/include/utils)
    target_include_directories(cube_pointer_validation PRIVATE "${GENERATED}")
    add_executable(cube_pointer_compile_probe matrix_compile.asc)
    target_compile_options(cube_pointer_compile_probe PRIVATE --npu-arch=dav-3510
        -iquote${REPO}/include -iquote${REPO} -iquote${REPO}/include/basic_api -iquote${REPO}/include/utils)
    target_include_directories(cube_pointer_compile_probe PRIVATE "${GENERATED}")
    return()
endif()
get_filename_component(SAMPLE_NAME "${SAMPLE_DIR}" NAME)
# The supplied note contains embedded comment terminators. Repair only that
# prose in the build copy, keeping the archived input and line count intact.
file(READ "${SAMPLE_DIR}/${SAMPLE_NAME}.asc" SAMPLE_TEXT)
string(REPLACE "Mmad*/Fixpipe*/LoadData*" "Mmad / Fixpipe / LoadData" SAMPLE_TEXT "${SAMPLE_TEXT}")
if(CUBE_FIX_BIAS)
    string(REPLACE "(__cc__ T*)bias2Local.GetPhyAddr()" "(__biasbuf__ T*)bias2Local.GetPhyAddr()" SAMPLE_TEXT "${SAMPLE_TEXT}")
endif()
if(CUBE_TENSOR_BASELINE)
    string(REGEX REPLACE "\\(__[a-z]+__ [A-Za-z0-9_]+\\*\\)([A-Za-z0-9_]+)\\.GetPhyAddr\\(\\)" "\\1" SAMPLE_TEXT "${SAMPLE_TEXT}")
endif()
if(SAMPLE_NAME STREQUAL "load_data_2dmx_l12l0")
    string(REPLACE "(__ca__ T*)a2Local.GetPhyAddr(), (__cb__ T*)b2Local.GetPhyAddr()"
        "(__ca__ TAL0Type*)a2Local.GetPhyAddr(), (__cb__ TBL0Type*)b2Local.GetPhyAddr()" SAMPLE_TEXT "${SAMPLE_TEXT}")
endif()
if(SAMPLE_NAME STREQUAL "mmad_load3dv2")
    string(REPLACE "(__ca__ U*)" "(__ca__ T*)" SAMPLE_TEXT "${SAMPLE_TEXT}")
    string(REPLACE "(__cb__ U*)" "(__cb__ T*)" SAMPLE_TEXT "${SAMPLE_TEXT}")
    string(REPLACE "(__cbuf__ U*)" "(__cbuf__ T*)" SAMPLE_TEXT "${SAMPLE_TEXT}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${SAMPLE_NAME}.asc" "${SAMPLE_TEXT}")
if(CUBE_PREPARED_SOURCE)
    configure_file("${CUBE_PREPARED_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/${SAMPLE_NAME}.asc" COPYONLY)
endif()
add_executable(demo "${CMAKE_CURRENT_BINARY_DIR}/${SAMPLE_NAME}.asc")
target_include_directories(demo PRIVATE "${SAMPLE_DIR}")
target_compile_definitions(demo PRIVATE SCENARIO_NUM=${SCENARIO_NUM} ASC_ARCHITECTURES=3510 ENABLE_UNITFLAG=1)
target_compile_options(demo PRIVATE --npu-arch=dav-3510
    -iquote${REPO}/include -iquote${REPO} -iquote${REPO}/include/basic_api
    -iquote${REPO}/include/utils)
target_include_directories(demo PRIVATE "${GENERATED}")
