### 7.1 原始形态示例：MatMul算子

```cpp
#include "basic_api/kernel_cube_intf.h"

class MatMulKernel {
public:
    __aicore__ void Init(GM_ADDR gmA, GM_ADDR gmB, GM_ADDR gmC) {
        // 分配存储空间
        pipe.InitBuffer(vecInQueL1, 1, 256 * 256 * sizeof(half));
        pipe.InitBuffer(vecInQueL0A, 1, 16 * 16 * sizeof(half));
        pipe.InitBuffer(vecInQueL0B, 1, 16 * 16 * sizeof(half));
        pipe.InitBuffer(vecOutQueL0C, 1, 16 * 16 * sizeof(float));
        
        // 设置GM Tensor
        gmA.SetGlobalBuffer((__gm__ half*)gmA, M * K);
        gmB.SetGlobalBuffer((__gm__ half*)gmB, K * N);
        gmC.SetGlobalBuffer((__gm__ float*)gmC, M * N);
    }
    
    __aicore__ void Process() {
        // 1. GM → L1搬运（ND转NZ）
        auto l1Tensor = vecInQueL1.AllocTensor<half>();
        
        Nd2NzParams nzParams;
        nzParams.ndNum = 1;
        nzParams.nValue = 16;
        nzParams.dValue = 16;
        // ... 其他参数
        
        DataCopy(l1Tensor, gmA, nzParams);
        vecInQueL1.EnQue(l1Tensor);
        
        // 2. L1 → L0A加载
        auto l1TensorA = vecInQueL1.DeQue<half>();
        auto l0ATensor = vecInQueL0A.AllocTensor<half>();
        
        LoadData2DParams loadParams;
        loadParams.startIndex = 0;
        loadParams.repeatTimes = 1;
        // ... 其他参数
        
        LoadData(l0ATensor, l1TensorA, loadParams);
        vecInQueL0A.EnQue(l0ATensor);
        
        // 3. L1 → L0B加载（类似）
        // ...
        
        // 4. Mmad矩阵乘加
        auto l0ATensor = vecInQueL0A.DeQue<half>();
        auto l0BTensor = vecInQueL0B.DeQue<half>();
        auto l0CTensor = vecOutQueL0C.AllocTensor<float>();
        
        MMadParams mmadParams;
        mmadParams.m = 16;
        mmadParams.k = 16;
        mmadParams.n = 16;
        mmadParams.cmatrixInitVal = 0;
        
        Mmad(l0CTensor, l0ATensor, l0BTensor, mmadParams);
        vecOutQueL0C.EnQue(l0CTensor);
        
        // 5. L0C → GM输出
        auto l0CTensor = vecOutQueL0C.DeQue<float>();
        
        FixpipeParams fixParams;
        fixParams.nValue = 16;
        fixParams.mValue = 16;
        fixParams.nz2ndEn = true;
        
        Fixpipe(gmC, l0CTensor, fixParams);
        
        // 释放资源
        vecInQueL1.FreeTensor(l1TensorA);
        vecInQueL0A.FreeTensor(l0ATensor);
        vecInQueL0B.FreeTensor(l0BTensor);
        vecOutQueL0C.FreeTensor(l0CTensor);
    }
    
private:
    TPipe pipe;
    TQue<QuePosition::L1, 1> vecInQueL1;
    TQue<QuePosition::L0A, 1> vecInQueL0A;
    TQue<QuePosition::L0B, 1> vecInQueL0B;
    TQue<QuePosition::L0C, 1> vecOutQueL0C;
    
    GlobalTensor<half> gmA;
    GlobalTensor<half> gmB;
    GlobalTensor<float> gmC;
};
```

### 6.2 扩展形态示例：MatMul算子

```cpp
#include "experimental/tensor_api/tensor.h"

using namespace AscendC::Te;

class MatMulKernel {
public:
    __aicore__ void Init(GM_ADDR gmA, GM_ADDR gmB, GM_ADDR gmC) {
        // 步骤1：创建GM Tensor（输入输出）
        // 使用ND Layout（行优先存储）
        gmTensorA = MakeTensor(MakeGMmemPtr<half>((__gm__ half*)gmA), MakeNDLayout<half>(M, K));
        gmTensorB = MakeTensor(MakeGMmemPtr<half>((__gm__ half*)gmB), MakeNDLayout<half>(K, N));
        gmTensorC = MakeTensor(MakeGMmemPtr<float>((__gm__ float*)gmC), MakeNDLayout<float>(M, N));
        
        // 步骤2：创建分块描述
        tile = MakeTile(MakeShape(16, 16));  // 16x16分块
    }
    
    __aicore__ void Process() {
        // 步骤1：创建内部Tensor
        // L1 Tensor使用NZ Layout（Cube优化格式）
        auto l1TensorA = MakeTensor(MakeL1memPtr<half>(l1AddrA), MakeNzLayout<half>(16, 16));
        auto l1TensorB = MakeTensor(MakeL1memPtr<half>(l1AddrB), MakeNzLayout<half>(16, 16));
        // L0 Tensor使用NZ Layout
        auto l0ATensor = MakeTensor(MakeL0AmemPtr<half>(l0AAddr), MakeNzLayout<half>(16, 16));
        auto l0BTensor = MakeTensor(MakeL0BmemPtr<half>(l0BAddr), MakeNzLayout<half>(16, 16));
        // L0C Tensor使用L0C Layout（结果格式）
        auto l0CTensor = MakeTensor(MakeL0CmemPtr<float>(l0CAddr), MakeL0CLayout<float>(16, 16));
        
        // 步骤2：循环分块计算
        for (int row = 0; row < M; row += 16) {
            for (int col = 0; col < N; col += 16) {
                for (int m = 0; m < K; m += 16) {
                    // 2.1 GM → L1搬运（Copy封装DataCopy，自动ND转NZ）
                    // 从GM的(row,m)位置提取16x16分块，搬运到L1
                    Copy(CopyGM2L1{}, l1TensorA, gmTensorA(MakeCoord(row, m), tile));
                    Copy(CopyGM2L1{}, l1TensorB, gmTensorB(MakeCoord(m, col), tile));
                    
                    // 2.2 L1 → L0搬运（Copy封装LoadData）
                    Copy(CopyL12L0A{}, l0ATensor, l1TensorA);
                    Copy(CopyL12L0B{}, l0BTensor, l1TensorB);
                    
                    // 2.3 Mmad矩阵乘加（Mad封装Mmad）
                    // Copy自动配置参数，计算16x16矩阵乘法
                    MmadParams params{16, 16, 16, 0, 0};
                    Mad(MmadOperation{}, l0CTensor, l0ATensor, l0BTensor, params);
                }
                
                // 2.4 L0C → GM输出（Copy封装Fixpipe）
                // 从L0C搬运到GM的(row,col)位置，自动格式转换
                Copy(CopyL0C2GM{}, gmTensorC(MakeCoord(row, col), tile), l0CTensor);
            }
        }
    }
    
private:
    // GM Tensor（输入输出）
    LocalTensor<GMPtr<half>, NDLayout<half>> gmTensorA;
    LocalTensor<GMPtr<half>, NDLayout<half>> gmTensorB;
    LocalTensor<GMPtr<float>, NDLayout<float>> gmTensorC;
    
    // 分块描述
    Tile<16, 16> tile;
};
```

**两种形态对比**：
- **原始形态**：手动配置参数（DataCopyParams、MMadParams），直接调用底层接口
- **扩展形态**：通过Layout和Copy/Mad封装，自动处理索引和格式转换
