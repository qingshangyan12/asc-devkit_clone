## 1. 向量加法样例

### 1.1 单核向量加法

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* input0;
extern __gm__ half* input1;
extern __gm__ half* output;
extern __gm__ uint32_t* total_length_ptr;

__aicore__ void vector_add_kernel()
{
    asc_init();
    
    uint32_t total_length = *total_length_ptr;
    uint32_t block_length = total_length / asc_get_block_num();
    uint32_t block_idx = asc_get_block_idx();
    
    __gm__ half* src0_gm = input0 + block_idx * block_length;
    __gm__ half* src1_gm = input1 + block_idx * block_length;
    __gm__ half* dst_gm = output + block_idx * block_length;
    
    __ubuf__ half src0[block_length];
    __ubuf__ half src1[block_length];
    __ubuf__ half dst[block_length];
    
    asc_copy_gm2ub((__ubuf__ void*)src0, (__gm__ void*)src0_gm, 
                   block_length * sizeof(half));
    asc_copy_gm2ub((__ubuf__ void*)src1, (__gm__ void*)src1_gm, 
                   block_length * sizeof(half));
    
    asc_sync_pipe(PIPE_MTE2);
    
    asc_add(dst, src0, src1, block_length);
    
    asc_sync_pipe(PIPE_V);
    
    asc_copy_ub2gm((__gm__ void*)dst_gm, (__ubuf__ void*)dst, 
                   block_length * sizeof(half));
    
    asc_sync_pipe(PIPE_MTE3);
}
```

### 1.2 带分块处理的向量加法

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* input0;
extern __gm__ half* input1;
extern __gm__ half* output;
extern __gm__ uint32_t* total_length_ptr;

__aicore__ void vector_add_blocked_kernel()
{
    asc_init();
    
    uint32_t total_length = *total_length_ptr;
    uint32_t block_idx = asc_get_block_idx();
    uint32_t block_num = asc_get_block_num();
    
    constexpr uint32_t ub_block_size = 1024;
    uint32_t my_length = total_length / block_num;
    uint32_t loop_count = (my_length + ub_block_size - 1) / ub_block_size;
    
    __ubuf__ half src0[ub_block_size];
    __ubuf__ half src1[ub_block_size];
    __ubuf__ half dst[ub_block_size];
    
    __gm__ half* src0_base = input0 + block_idx * my_length;
    __gm__ half* src1_base = input1 + block_idx * my_length;
    __gm__ half* dst_base = output + block_idx * my_length;
    
    for (uint32_t i = 0; i < loop_count; ++i) {
        uint32_t offset = i * ub_block_size;
        uint32_t cur_size = (i == loop_count - 1) ? 
            (my_length - offset) : ub_block_size;
        
        asc_copy_gm2ub((__ubuf__ void*)src0, 
                       (__gm__ void*)(src0_base + offset), 
                       cur_size * sizeof(half));
        asc_copy_gm2ub((__ubuf__ void*)src1, 
                       (__gm__ void*)(src1_base + offset), 
                       cur_size * sizeof(half));
        
        asc_sync_pipe(PIPE_MTE2);
        
        asc_add(dst, src0, src1, cur_size);
        
        asc_sync_pipe(PIPE_V);
        
        asc_copy_ub2gm((__gm__ void*)(dst_base + offset), 
                       (__ubuf__ void*)dst, 
                       cur_size * sizeof(half));
        
        asc_sync_pipe(PIPE_MTE3);
    }
}
```

## 2. 矩阵乘法样例

### 2.1 基本矩阵乘法

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* matrix_a;
extern __gm__ half* matrix_b;
extern __gm__ float* matrix_c;
extern __gm__ uint16_t* m_dim_ptr;
extern __gm__ uint16_t* k_dim_ptr;
extern __gm__ uint16_t* n_dim_ptr;

__aicore__ void matmul_kernel()
{
    asc_init();
    
    uint16_t M = *m_dim_ptr;
    uint16_t K = *k_dim_ptr;
    uint16_t N = *n_dim_ptr;
    
    constexpr uint32_t l1_size = 4096;
    constexpr uint32_t l0_size = 512;
    constexpr uint32_t l0c_size = 256;
    
    __cbuf__ half a_l1[l1_size];
    __cbuf__ half b_l1[l1_size];
    __ca__ half a_l0a[l0_size];
    __cb__ half b_l0b[l0_size];
    __cc__ float c_l0c[l0c_size];
    
    asc_copy_gm2l1((__cbuf__ void*)a_l1, (__gm__ void*)matrix_a, 
                   M * K * sizeof(half));
    asc_copy_gm2l1((__cbuf__ void*)b_l1, (__gm__ void*)matrix_b, 
                   K * N * sizeof(half));
    
    asc_sync_pipe(PIPE_MTE2);
    
    asc_copy_l12l0a((__ca__ void*)a_l0a, (__cbuf__ void*)a_l1, 0, 1, 1, 0);
    asc_copy_l12l0b((__cb__ void*)b_l0b, (__cbuf__ void*)b_l1, 0, 1, 1, 0);
    
    asc_sync_pipe(PIPE_MTE1);
    
    asc_mmad((__cc__ float*)c_l0c, (__ca__ half*)a_l0a, (__cb__ half*)b_l0b,
             M, K, N, 0, false, false, true);
    
    asc_sync_pipe(PIPE_M);
    
    asc_copy_l0c2gm((__gm__ void*)matrix_c, (__cc__ void*)c_l0c, 
                    M * N * sizeof(float));
    
    asc_sync_pipe(PIPE_MTE3);
}
```

### 2.2 分块矩阵乘法

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* matrix_a;
extern __gm__ half* matrix_b;
extern __gm__ float* matrix_c;
extern __gm__ uint16_t* m_dim_ptr;
extern __gm__ uint16_t* k_dim_ptr;
extern __gm__ uint16_t* n_dim_ptr;

__aicore__ void matmul_blocked_kernel()
{
    asc_init();
    
    uint16_t M = *m_dim_ptr;
    uint16_t K = *k_dim_ptr;
    uint16_t N = *n_dim_ptr;
    
    constexpr uint16_t BLOCK_M = 16;
    constexpr uint16_t BLOCK_K = 16;
    constexpr uint16_t BLOCK_N = 16;
    
    constexpr uint32_t l1_size = BLOCK_M * BLOCK_K + BLOCK_K * BLOCK_N;
    constexpr uint32_t l0a_size = BLOCK_M * BLOCK_K;
    constexpr uint32_t l0b_size = BLOCK_K * BLOCK_N;
    constexpr uint32_t l0c_size = BLOCK_M * BLOCK_N;
    
    __cbuf__ half a_l1[l0a_size];
    __cbuf__ half b_l1[l0b_size];
    __ca__ half a_l0a[l0a_size];
    __cb__ half b_l0b[l0b_size];
    __cc__ float c_l0c[l0c_size];
    
    for (uint16_t m_block = 0; m_block < M / BLOCK_M; ++m_block) {
        for (uint16_t n_block = 0; n_block < N / BLOCK_N; ++n_block) {
            for (uint16_t k_block = 0; k_block < K / BLOCK_K; ++k_block) {
                asc_copy_gm2l1((__cbuf__ void*)a_l1, 
                    (__gm__ void*)(matrix_a + (m_block * BLOCK_M * K + k_block * BLOCK_K)),
                    BLOCK_M * BLOCK_K * sizeof(half));
                asc_copy_gm2l1((__cbuf__ void*)b_l1, 
                    (__gm__ void*)(matrix_b + (k_block * BLOCK_K * N + n_block * BLOCK_N)),
                    BLOCK_K * BLOCK_N * sizeof(half));
                
                asc_sync_pipe(PIPE_MTE2);
                
                asc_copy_l12l0a((__ca__ void*)a_l0a, (__cbuf__ void*)a_l1, 
                               0, 1, 1, 0);
                asc_copy_l12l0b((__cb__ void*)b_l0b, (__cbuf__ void*)b_l1, 
                               0, 1, 1, 0);
                
                asc_sync_pipe(PIPE_MTE1);
                
                asc_mmad((__cc__ float*)c_l0c, (__ca__ half*)a_l0a, 
                         (__cb__ half*)b_l0b, BLOCK_M, BLOCK_K, BLOCK_N,
                         0, false, (k_block > 0), false);
                
                asc_sync_pipe(PIPE_M);
            }
            
            asc_copy_l0c2gm((__gm__ void*)(matrix_c + (m_block * BLOCK_M * N + n_block * BLOCK_N)),
                           (__cc__ void*)c_l0c, BLOCK_M * BLOCK_N * sizeof(float));
            
            asc_sync_pipe(PIPE_MTE3);
        }
    }
}
```

## 3. Softmax 样例

### 3.1 Softmax 计算流程

```
1. 计算 exp(x)
2. 归约求 sum(exp(x))
3. 计算 exp(x) / sum
```

### 3.2 完整代码

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* input;
extern __gm__ half* output;
extern __gm__ uint32_t* length_ptr;

__aicore__ void softmax_kernel()
{
    asc_init();
    
    uint32_t length = *length_ptr;
    
    __ubuf__ half src[length];
    __ubuf__ half exp_result[length];
    __ubuf__ half sum_result[1];
    __ubuf__ half dst[length];
    
    asc_copy_gm2ub((__ubuf__ void*)src, (__gm__ void*)input, 
                   length * sizeof(half));
    
    asc_sync_pipe(PIPE_MTE2);
    
    asc_exp((__ubuf__ half*)exp_result, (__ubuf__ half*)src, length);
    
    asc_sync_pipe(PIPE_V);
    
    asc_repeat_reduce_sum((__ubuf__ half*)sum_result, 
                          (__ubuf__ half*)exp_result, 1);
    
    asc_sync_pipe(PIPE_V);
    
    asc_duplicate((__ubuf__ half*)dst, (__ubuf__ half*)sum_result, length);
    
    asc_sync_pipe(PIPE_V);
    
    asc_div((__ubuf__ half*)dst, (__ubuf__ half*)exp_result, 
            (__ubuf__ half*)dst, length);
    
    asc_sync_pipe(PIPE_V);
    
    asc_copy_ub2gm((__gm__ void*)output, (__ubuf__ void*)dst, 
                   length * sizeof(half));
    
    asc_sync_pipe(PIPE_MTE3);
}
```

### 3.3 安全 Softmax（数值稳定）

## 4. 多核协同计算样例

### 4.1 多核归约求和

```cpp
#include "c_api/asc_simd.h"

extern __gm__ float* input;
extern __gm__ float* output;
extern __gm__ uint32_t* total_length_ptr;
extern __gm__ uint64_t* ffts_addr_ptr;

__aicore__ void multi_core_reduce_kernel()
{
    asc_init();
    
    uint64_t ffts_addr = *ffts_addr_ptr;
    asc_set_ffts_base_addr(ffts_addr);
    
    int64_t block_idx = asc_get_block_idx();
    int64_t block_num = asc_get_block_num();
    
    uint32_t total_length = *total_length_ptr;
    uint32_t block_length = total_length / block_num;
    
    __gm__ float* src_gm = input + block_idx * block_length;
    
    __ubuf__ float src[block_length];
    __ubuf__ float partial_sum[1];
    
    asc_copy_gm2ub((__ubuf__ void*)src, (__gm__ void*)src_gm, 
                   block_length * sizeof(float));
    
    asc_sync_pipe(PIPE_MTE2);
    
    asc_repeat_reduce_sum((__ubuf__ float*)partial_sum, 
                          (__ubuf__ float*)src, 1);
    
    asc_sync_pipe(PIPE_V);
    
    int64_t flag_id = 1;
    
    asc_set_atomic_add_float();
    asc_copy_ub2gm((__gm__ void*)output, (__ubuf__ void*)partial_sum, 
                   sizeof(float));
    asc_set_atomic_none();
    
    asc_sync_pipe(PIPE_MTE3);
    
    asc_sync_block_arrive(PIPE_S, flag_id);
    asc_sync_block_wait(PIPE_S, flag_id);
}
```

## 9. 性能优化样例

### 9.1 双缓冲优化

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* input;
extern __gm__ half* output;
extern __gm__ uint32_t* total_length_ptr;

__aicore__ void double_buffer_kernel()
{
    asc_init();
    
    uint32_t total_length = *total_length_ptr;
    constexpr uint32_t buffer_size = 512;
    
    __ubuf__ half buffer0[buffer_size];
    __ubuf__ half buffer1[buffer_size];
    __ubuf__ half result0[buffer_size];
    __ubuf__ half result1[buffer_size];
    
    uint32_t loop_count = total_length / buffer_size;
    
    for (uint32_t i = 0; i < loop_count; ++i) {
        if (i % 2 == 0) {
            asc_copy_gm2ub((__ubuf__ void*)buffer0, 
                           (__gm__ void*)(input + i * buffer_size), 
                           buffer_size * sizeof(half));
        } else {
            asc_copy_gm2ub((__ubuf__ void*)buffer1, 
                           (__gm__ void*)(input + i * buffer_size), 
                           buffer_size * sizeof(half));
        }
        
        asc_sync_pipe(PIPE_MTE2);
        
        if (i % 2 == 0) {
            asc_add(result0, buffer0, buffer0, buffer_size);
        } else {
            asc_add(result1, buffer1, buffer1, buffer_size);
        }
        
        asc_sync_pipe(PIPE_V);
        
        if (i > 0) {
            if ((i - 1) % 2 == 0) {
                asc_copy_ub2gm((__gm__ void*)(output + (i - 1) * buffer_size), 
                               (__ubuf__ void*)result0, buffer_size * sizeof(half));
            } else {
                asc_copy_ub2gm((__gm__ void*)(output + (i - 1) * buffer_size), 
                               (__ubuf__ void*)result1, buffer_size * sizeof(half));
            }
            asc_sync_pipe(PIPE_MTE3);
        }
    }
    
    if ((loop_count - 1) % 2 == 0) {
        asc_copy_ub2gm((__gm__ void*)(output + (loop_count - 1) * buffer_size), 
                       (__ubuf__ void*)result0, buffer_size * sizeof(half));
    } else {
        asc_copy_ub2gm((__gm__ void*)(output + (loop_count - 1) * buffer_size), 
                       (__ubuf__ void*)result1, buffer_size * sizeof(half));
    }
    asc_sync_pipe(PIPE_MTE3);
}
```

```cpp
#include "c_api/asc_simd.h"

extern __gm__ half* input;

    asc_sync_pipe(PIPE_V);

                   length * sizeof(half));
```
