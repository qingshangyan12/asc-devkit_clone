# 原始形态示例：MatMul算子（完整参数版）

```cpp
#include "basic_api/kernel_cube_intf.h"

// 定义矩阵维度（假设16x16小矩阵）
constexpr int32_t M = 16;  // 矩阵A的行数
constexpr int32_t K = 16;  // 矩阵A的列数=矩阵B的行数
constexpr int32_t N = 16;  // 矩阵B的列数

class MatMulKernel {
public:
    __aicore__ void Init(GM_ADDR gmA, GM_ADDR gmB, GM_ADDR gmC) {
        // 步骤1：初始化管道
        pipe.Init();
        
        // 步骤2：分配存储空间
        // L1：缓存矩阵A和矩阵B的NZ格式数据
        pipe.InitBuffer(vecInQueL1, 2, 16 * 16 * sizeof(half));  // 2块，每块256元素
        
        // L0A/L0B：存储16x16矩阵块（NZ格式）
        pipe.InitBuffer(vecInQueL0A, 1, 16 * 16 * sizeof(half));
        pipe.InitBuffer(vecInQueL0B, 1, 16 * 16 * sizeof(half));
        
        // L0C：存储矩阵乘加结果（float累加）
        pipe.InitBuffer(vecOutQueL0C, 1, 16 * 16 * sizeof(float));
        
        // 步骤3：设置GM Tensor地址
        gmA.SetGlobalBuffer((__gm__ half*)gmA, M * K);
        gmB.SetGlobalBuffer((__gm__ half*)gmB, K * N);
        gmC.SetGlobalBuffer((__gm__ float*)gmC, M * N);
    }
    
    __aicore__ void Process() {
        // ========== 步骤1：GM → L1搬运（矩阵A，ND转NZ）==========
        auto l1TensorA = vecInQueL1.AllocTensor<half>();
        
        // Nd2NzParams完整参数配置（ND转NZ）
        Nd2NzParams nzParamsA;
        nzParamsA.ndNum = 1;                  // ND矩阵数量：1个
        nzParamsA.nValue = M;                 // N维度：16行
        nzParamsA.dValue = K;                 // D维度：16列
        nzParamsA.srcNdMatrixStride = 0;      // ND矩阵间间隔：0（单矩阵）
        nzParamsA.srcDValue = K;              // ND矩阵的D值：16
        nzParamsA.dstNzC0Stride = K;          // NZ格式的C0步长：16
        nzParamsA.dstNzNStride = K * 16;      // NZ格式的N步长：256
        nzParamsA.dstNzMatrixStride = M * K;  // NZ矩阵总步长：256
        
        DataCopy(l1TensorA, gmA, nzParamsA);  // 执行搬运+格式转换
        vecInQueL1.EnQue(l1TensorA);
        
        // ========== 步骤2：GM → L1搬运（矩阵B，ND转NZ）==========
        auto l1TensorB = vecInQueL1.AllocTensor<half>();
        
        // Nd2NzParams完整参数配置（矩阵B）
        Nd2NzParams nzParamsB;
        nzParamsB.ndNum = 1;
        nzParamsB.nValue = K;
        nzParamsB.dValue = N;
        nzParamsB.srcNdMatrixStride = 0;
        nzParamsB.srcDValue = N;
        nzParamsB.dstNzC0Stride = N;
        nzParamsB.dstNzNStride = N * 16;
        nzParamsB.dstNzMatrixStride = K * N;
        
        DataCopy(l1TensorB, gmB, nzParamsB);
        vecInQueL1.EnQue(l1TensorB);
        
        // ========== 步骤3：L1 → L0A加载（矩阵A）==========
        auto l1TensorA_deque = vecInQueL1.DeQue<half>();
        auto l0ATensor = vecInQueL0A.AllocTensor<half>();
        
        // LoadData2DParams完整参数配置
        LoadData2DParams loadParamsA;
        loadParamsA.startIndex = 0;           // 起始索引：0
        loadParamsA.repeatTimes = 1;          // 重复次数：1
        loadParamsA.srcStride = 16;           // 源步长：16（NZ格式的分块大小）
        loadParamsA.sid = 0;                  // SID标识：0
        loadParamsA.dstGap = 0;               // 目标间隔：0
        loadParamsA.ifTranspose = false;      // 是否转置：否
        loadParamsA.enRepeatSize = 16;        // 每次重复处理的元素数：16
        loadParamsA.enRepeatStride = 16;      // 重复步长：16
        
        LoadData(l0ATensor, l1TensorA_deque, loadParamsA);
        vecInQueL0A.EnQue(l0ATensor);
        vecInQueL1.FreeTensor(l1TensorA_deque);
        
        // ========== 步骤4：L1 → L0B加载（矩阵B）==========
        auto l1TensorB_deque = vecInQueL1.DeQue<half>();
        auto l0BTensor = vecInQueL0B.AllocTensor<half>();
        
        // LoadData2DParams完整参数配置（矩阵B）
        LoadData2DParams loadParamsB;
        loadParamsB.startIndex = 0;
        loadParamsB.repeatTimes = 1;
        loadParamsB.srcStride = 16;
        loadParamsB.sid = 0;
        loadParamsB.dstGap = 0;
        loadParamsB.ifTranspose = false;
        loadParamsB.enRepeatSize = 16;
        loadParamsB.enRepeatStride = 16;
        
        LoadData(l0BTensor, l1TensorB_deque, loadParamsB);
        vecInQueL0B.EnQue(l0BTensor);
        vecInQueL1.FreeTensor(l1TensorB_deque);
        
        // ========== 步骤5：Mmad矩阵乘加 ==========
        auto l0ATensor_deque = vecInQueL0A.DeQue<half>();
        auto l0BTensor_deque = vecInQueL0B.DeQue<half>();
        auto l0CTensor = vecOutQueL0C.AllocTensor<float>();
        
        // MMadParams完整参数配置
        MMadParams mmadParams;
        mmadParams.m = M;                     // M维度：16行
        mmadParams.k = K;                     // K维度：16（A的列=B的行）
        mmadParams.n = N;                     // N维度：16列
        mmadParams.unitFlag = 0;              // 计算模式：标准模式
        mmadParams.cmatrixInitVal = 0;        // C矩阵初始值：从零累加
        
        // 执行矩阵乘加：l0CTensor = l0ATensor * l0BTensor
        Mmad(l0CTensor, l0ATensor_deque, l0BTensor_deque, mmadParams);
        vecOutQueL0C.EnQue(l0CTensor);
        vecInQueL0A.FreeTensor(l0ATensor_deque);
        vecInQueL0B.FreeTensor(l0BTensor_deque);
        
        // ========== 步骤6：L0C → GM输出（NZ转ND）==========
        auto l0CTensor_deque = vecOutQueL0C.DeQue<float>();
        
        // FixpipeParams完整参数配置
        FixpipeParams fixParams;
        fixParams.nValue = N;                 // N维度：16
        fixParams.mValue = M;                 // M维度：16
        fixParams.kValue = K;                 // K维度：16
        fixParams.nz2ndEn = true;             // NZ转ND：开启
        fixParams.srcGap = 0;                 // 源间隔：0
        fixParams.dstGap = 0;                 // 目标间隔：0
        fixParams.sid = 0;                    // SID标识：0
        fixParams.quantPre = QuantPre::NONE;  //量化预处理：无
        
        // 执行输出+格式转换
        Fixpipe(gmC, l0CTensor_deque, fixParams);
        vecOutQueL0C.FreeTensor(l0CTensor_deque);
    }
    
private:
    // 管道和队列
    TPipe pipe;
    TQue<QuePosition::L1, 2> vecInQueL1;     // L1队列，深度2
    TQue<QuePosition::L0A, 1> vecInQueL0A;   // L0A队列，深度1
    TQue<QuePosition::L0B, 1> vecInQueL0B;   // L0B队列，深度1
    TQue<QuePosition::L0C, 1> vecOutQueL0C;  // L0C队列，深度1
    
    // GM Tensor
    GlobalTensor<half> gmA;                  // 矩阵A
    GlobalTensor<half> gmB;                  // 矩阵B
    GlobalTensor<float> gmC;                 // 结果矩阵C
};
```

# 扩展形态示例：MatMul算子（完整参数版）

```cpp
#include "experimental/tensor_api/tensor.h"

using namespace AscendC::Te;

// 定义矩阵维度（假设128x128大矩阵，分块为16x16）
constexpr int32_t M = 128;  // 矩阵A的行数
constexpr int32_t K = 128;  // 矩阵A的列数=矩阵B的行数
constexpr int32_t N = 128;  // 矩阵B的列数
constexpr int32_t BLOCK_SIZE = 16;  // 分块大小

class MatMulKernel {
public:
    __aicore__ void Init(GM_ADDR gmA, GM_ADDR gmB, GM_ADDR gmC) {
        // 步骤1：分配存储地址
        l1AddrA = asc_get_phy_buf_addr(0);                    // L1地址A（矩阵A缓存）
        l1AddrB = asc_get_phy_buf_addr(BLOCK_SIZE * BLOCK_SIZE * sizeof(half));  // L1地址B
        l0AAddr = asc_get_phy_buf_addr(2 * BLOCK_SIZE * BLOCK_SIZE * sizeof(half));  // L0A地址
        l0BAddr = asc_get_phy_buf_addr(l0AAddr + BLOCK_SIZE * BLOCK_SIZE * sizeof(half));  // L0B地址
        l0CAddr = asc_get_phy_buf_addr(l0BAddr + BLOCK_SIZE * BLOCK_SIZE * sizeof(half));  // L0C地址
        
        // 步骤2：创建GM Tensor（使用ND Layout，行优先存储）
        gmTensorA = MakeTensor(
            MakeGMmemPtr<half>((__gm__ half*)gmA),
            MakeNDLayout<half>(M, K)             // ND Layout：MxK
        );
        gmTensorB = MakeTensor(
            MakeGMmemPtr<half>((__gm__ half*)gmB),
            MakeNDLayout<half>(K, N)             // ND Layout：KxN
        );
        gmTensorC = MakeTensor(
            MakeGMmemPtr<float>((__gm__ float*)gmC),
            MakeNDLayout<float>(M, N)            // ND Layout：MxN
        );
        
        // 步骤3：创建分块描述
        tile = MakeTile(MakeShape(BLOCK_SIZE, BLOCK_SIZE));  // 16x16分块
    }
    
    __aicore__ void Process() {
        // 步骤1：创建内部Tensor
        // L1 Tensor使用NZ Layout（Cube优化格式）
        auto l1TensorA = MakeTensor(
            MakeL1memPtr<half>(l1AddrA),
            MakeNzLayout<half>(BLOCK_SIZE, BLOCK_SIZE)
        );
        auto l1TensorB = MakeTensor(
            MakeL1memPtr<half>(l1AddrB),
            MakeNzLayout<half>(BLOCK_SIZE, BLOCK_SIZE)
        );
        
        // L0 Tensor使用NZ Layout
        auto l0ATensor = MakeTensor(
            MakeL0AmemPtr<half>(l0AAddr),
            MakeNzLayout<half>(BLOCK_SIZE, BLOCK_SIZE)
        );
        auto l0BTensor = MakeTensor(
            MakeL0BmemPtr<half>(l0BAddr),
            MakeNzLayout<half>(BLOCK_SIZE, BLOCK_SIZE)
        );
        
        // L0C Tensor使用L0C Layout（结果格式）
        auto l0CTensor = MakeTensor(
            MakeL0CmemPtr<float>(l0CAddr),
            MakeL0CLayout<float>(BLOCK_SIZE, BLOCK_SIZE)
        );
        
        // 步骤2：循环分块计算
        for (int32_t row = 0; row < M; row += BLOCK_SIZE) {
            for (int32_t col = 0; col < N; col += BLOCK_SIZE) {
                // 初始化累加标志（第一次K分块从零开始）
                bool isFirstKBlock = true;
                
                for (int32_t k = 0; k < K; k += BLOCK_SIZE) {
                    // ========== 2.1 GM → L1搬运（Copy封装DataCopy，自动ND转NZ）==========
                    // 从GM的(row,k)位置提取16x16分块，搬运到L1
                    // Copy自动配置Nd2NzParams参数
                    Copy(CopyGM2L1{}, l1TensorA, gmTensorA(MakeCoord(row, k), tile));
                    Copy(CopyGM2L1{}, l1TensorB, gmTensorB(MakeCoord(k, col), tile));
                    
                    // ========== 2.2 L1 → L0搬运（Copy封装LoadData）==========
                    // Copy自动配置LoadData2DParams参数
                    Copy(CopyL12L0A{}, l0ATensor, l1TensorA);
                    Copy(CopyL12L0B{}, l0BTensor, l1TensorB);
                    
                    // ========== 2.3 Mmad矩阵乘加（Mad封装Mmad）==========
                    // MadParams完整参数配置
                    MmadParams params;
                    params.m = BLOCK_SIZE;                // M维度：16行
                    params.k = BLOCK_SIZE;                // K维度：16
                    params.n = BLOCK_SIZE;                // N维度：16列
                    params.unitFlag = 0;                  // 标准计算模式
                    params.cmatrixInitVal = isFirstKBlock ? 0 : 1;  // 第一次从零累加，后续累加
                    
                    // Mad自动调用底层Mmad
                    // Copy自动配置参数，计算16x16矩阵乘法
                    Mad(MmadOperation{}, l0CTensor, l0ATensor, l0BTensor, params);
                    
                    isFirstKBlock = false;  // 后续K分块累加
                }
                
                // ========== 2.4 L0C → GM输出（Copy封装Fixpipe）==========
                // 从L0C搬运到GM的(row,col)位置，自动格式转换（L0C转ND）
                // Copy自动配置FixpipeParams参数
                Copy(CopyL0C2GM{}, gmTensorC(MakeCoord(row, col), tile), l0CTensor);
            }
        }
    }
    
private:
    // GM Tensor（输入输出）
    LocalTensor<TensorAttribute<GMPtr<half>, NDLayout<half>>> gmTensorA;
    LocalTensor<TensorAttribute<GMPtr<half>, NDLayout<half>>> gmTensorB;
    LocalTensor<TensorAttribute<GMPtr<float>, NDLayout<float>>> gmTensorC;
    
    // 分块描述
    Tile<BLOCK_SIZE, BLOCK_SIZE> tile;
    
    // 存储地址
    uint32_t l1AddrA, l1AddrB;
    uint32_t l0AAddr, l0BAddr, l0CAddr;
};
```

# RegTensor示例：向量加法算子（完整参数版）

```cpp
#include "basic_api/reg_compute/kernel_reg_compute_intf.h"

using namespace AscendC::Reg;

// 定义向量长度（128个half元素，单次迭代）
constexpr int32_t LENGTH = 128;

class VectorAddKernel {
public:
    __aicore__ void Process(__ubuf__ half* ubSrc0, __ubuf__ half* ubSrc1, __ubuf__ half* ubDst) {
        // ========== 步骤1：创建RegTensor变量 ==========
        RegTensor<half> regSrc0;       // Reg寄存器，存储源数据0
        RegTensor<half> regSrc1;       // Reg寄存器，存储源数据1
        RegTensor<half> regDst;        // Reg寄存器，存储结果
        MaskReg mask;                  // Mask寄存器（掩码）
        
        // ========== 步骤2：创建Mask（完整参数）==========
        // CreateMask完整参数：
        // - T：数据类型（half）
        // - MaskPattern：掩码模式（ALL表示所有元素参与）
        mask = CreateMask<half, MaskPattern::ALL>();
        
        // ========== 步骤3：UB → Reg搬运（LoadAlign完整参数）==========
        // LoadAlign：对齐搬运，UB → Reg
        // 参数：
        // - dstReg：目标RegTensor
        // - srcAddr：UB源地址（必须32Byte对齐）
        LoadAlign<half>(regSrc0, ubSrc0);  // UB → Reg（源数据0）
        LoadAlign<half>(regSrc1, ubSrc1);  // UB → Reg（源数据1）
        
        // ========== 步骤4：在Reg中计算（Add完整参数）==========
        // Add：Reg中加法
        // 参数：
        // - dst：目标RegTensor
        // - src0：源RegTensor0
        // - src1：源RegTensor1
        // - mask：掩码（控制哪些元素参与）
        // - 可选：MaskMergeMode（ZEROING/KEEPING）
        Add<half>(regDst, regSrc0, regSrc1, mask);  // regDst = regSrc0 + regSrc1
        
        // ========== 步骤5：Reg → UB搬运（StoreAlign完整参数）==========
        // StoreAlign：对齐搬运，Reg → UB
        // 参数：
        // - dstAddr：UB目标地址（必须32Byte对齐）
        // - srcReg：源RegTensor
        // - mask：掩码
        StoreAlign<half>(ubDst, regDst, mask);  // Reg → UB（结果）
    }
};

// 算子入口
extern __aicore__ void vector_add_reg(__ubuf__ half* src0, __ubuf__ half* src1, __ubuf__ half* dst) {
    VectorAddKernel op;
    op.Process(src0, src1, dst);
}
```

# RegTensor示例：多步计算（反复使用同一数据）

```cpp
#include "basic_api/reg_compute/kernel_reg_compute_intf.h"

using namespace AscendC::Reg;

constexpr int32_t LENGTH = 128;

class MultiStepComputeKernel {
public:
    __aicore__ void Process(__ubuf__ half* ubSrc, 
                           __ubuf__ half* ubDstAdd, 
                           __ubuf__ half* ubDstMul, 
                           __ubuf__ half* ubDstRelu) {
        // ========== 步骤1：创建RegTensor变量 ==========
        RegTensor<half> regSrc;        // 源数据Reg
        RegTensor<half> regDstAdd;     // 加法结果Reg
        RegTensor<half> regDstMul;     // 乘法结果Reg
        RegTensor<half> regDstRelu;    // ReLU结果Reg
        RegTensor<half> regTemp;       // 临时Reg（用于乘法）
        MaskReg mask = CreateMask<half, MaskPattern::ALL>();
        
        // ========== 步骤2：只搬运一次到Reg ==========
        LoadAlign<half>(regSrc, ubSrc);  // UB → Reg（只搬运一次）
        
        // ========== 步骤3：在Reg中反复计算（无需再次搬运）==========
        
        // 计算1：加法（regSrc + regSrc）
        Add<half>(regDstAdd, regSrc, regSrc, mask);
        
        // 计算2：乘法（regSrc * regSrc）
        Mul<half>(regDstMul, regSrc, regSrc, mask);
        
        // 计算3：ReLU激活（abs(regSrc)如果regSrc<0）
        Relu<half>(regDstRelu, regSrc, mask);
        
        // 计算4：归约求和（Reduce）
        RegTensor<half> regSum;
        Reduce<ReduceType::SUM, half>(regSum, regSrc, mask);
        
        // ========== 步骤4：一次性搬运回UB ==========
        StoreAlign<half>(ubDstAdd, regDstAdd, mask);
        StoreAlign<half>(ubDstMul, regDstMul, mask);
        StoreAlign<half>(ubDstRelu, regDstRelu, mask);
        
        // 优势：减少了UB访问次数（从UB读取1次，写入3次）
        // 对比：如果用LocalTensor，需要从UB读取3次（加法、乘法、ReLU各读一次）
    }
};
```

# 三种形态对比总结

| 特性 | 原始形态 | 扩展形态 | RegTensor |
|------|---------|---------|-----------|
| **参数配置** | 手动配置完整参数（DataCopyParams、MMadParams等） | Layout自动配置，Copy/Mad封装 | 参数简洁（mask控制） |
| **代码复杂度** | 高（需理解所有参数） | 中（Layout抽象简化） | 低（只需搬运和mask） |
| **格式转换** | 手动配置Nd2NzParams | Copy自动处理 | 无格式概念 |
| **性能优化** | 精细控制每个参数 | 抽象层有一定开销 | 数据复用，减少UB访问 |
| **适用场景** | 性能极限优化 | 复杂算子、矩阵计算 | 反复计算同一数据 |
