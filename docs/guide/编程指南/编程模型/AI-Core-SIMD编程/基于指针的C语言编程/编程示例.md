## Cube算子示例：矩阵乘法完整实现

矩阵乘法是典型的Cube算子，展示L1/L0内存层级与矩阵计算接口的使用方式。

### 算子功能描述

输入矩阵A（M×K）和矩阵B（K×N），计算输出矩阵C = A × B。

### 完整代码实现

```cpp
extern "C" __global__ __aicore__ void matmul_kernel(__gm__ half* a_gm, __gm__ half* b_gm, __gm__ float* c_gm, uint32_t M, uint32_t K, uint32_t N)
{
    // ===== 第一步：内存规划与地址分配 =====
    constexpr uint32_t BLOCK_M = 128;  // 分块M维度
    constexpr uint32_t BLOCK_K = 64;   // 分块K维度
    constexpr uint32_t BLOCK_N = 64;   // 分块N维度
    
    __cbuf__ half a_l1[BLOCK_M * BLOCK_K];   // L1中转区
    __cbuf__ half b_l1[BLOCK_K * BLOCK_N];   // L1中转区
    __ca__ half a_l0a[BLOCK_M * BLOCK_K];    // L0A输入
    __cb__ half b_l0b[BLOCK_K * BLOCK_N];    // L0B输入
    __cc__ float c_l0c[BLOCK_M * BLOCK_N];   // L0C输出
    
    // ===== 第二步：数据搬入 =====
    asc_copy_gm2l1(a_l1, a_gm, BLOCK_M * BLOCK_K);  // GM→L1
    asc_copy_gm2l1(b_l1, b_gm, BLOCK_K * BLOCK_N);  // GM→L1
    asc_sync_notify(PIPE_MTE2, PIPE_MTE1, EVENT_ID0);  // MTE2通知MTE1
    asc_sync_wait(PIPE_MTE2, PIPE_MTE1, EVENT_ID0);    // MTE1等待MTE2
    
    // ===== L1→L0A/L0B =====
    asc_copy_l12l0a(a_l0a, a_l1, 0, 1, 1, 0);  // L1→L0A
    asc_copy_l12l0b(b_l0b, b_l1, 0, 1, 1, 0);  // L1→L0B
    asc_sync_notify(PIPE_MTE1, PIPE_CUBE, EVENT_ID0);  // MTE1通知Cube
    asc_sync_wait(PIPE_MTE1, PIPE_CUBE, EVENT_ID0);    // Cube等待MTE1
    
    // ===== 第三步：计算执行 =====
    asc_mmad(c_l0c, a_l0a, b_l0b, BLOCK_M, BLOCK_K, BLOCK_N, 0, false, false, false);  // 矩阵乘
    asc_sync_notify(PIPE_CUBE, PIPE_MTE3, EVENT_ID0);  // Cube通知MTE3
    asc_sync_wait(PIPE_CUBE, PIPE_MTE3, EVENT_ID0);    // MTE3等待Cube
    
    // ===== 第四步：结果搬出 =====
    asc_copy_l0c2gm(c_gm, c_l0c, BLOCK_M * BLOCK_N);  // L0C→GM
}
```

## Mix算子示例：Matmul和LeakyRelu融合计算

实现Matmul和LeakyRelu融合计算，展示Cube单元和Vector单元协同计算的编程模式。

### 算子功能描述

实现Matmul和LeakyRelu融合计算，计算公式如下：

Matmul计算：C = A x B
LeakyRelu计算: C >= 0 ? C : (C x 0.001)

### 完整代码实现

```cpp
extern "C" __global__ __aicore__ void matmul_kernel(__gm__ half* a_gm, __gm__ half* b_gm, __gm__ float* c_gm, uint32_t M, uint32_t K, uint32_t N)
{
    constexpr uint32_t BLOCK_M = 128;  // 分块M维度
    constexpr uint32_t BLOCK_K = 64;   // 分块K维度
    constexpr uint32_t BLOCK_N = 64;   // 分块N维度

    if ASC_IS_AIC {
        __cbuf__ half a_l1[BLOCK_M * BLOCK_K];   // L1中转区
        __cbuf__ half b_l1[BLOCK_K * BLOCK_N];   // L1中转区
        __ca__ half a_l0a[BLOCK_M * BLOCK_K];    // L0A输入
        __cb__ half b_l0b[BLOCK_K * BLOCK_N];    // L0B输入
        __cc__ float c_l0c[BLOCK_M * BLOCK_N];   // L0C输出
        
        // ===== 第二步：数据搬入 =====
        asc_copy_gm2l1(a_l1, a_gm, BLOCK_M * BLOCK_K);  // GM→L1
        asc_copy_gm2l1(b_l1, b_gm, BLOCK_K * BLOCK_N);  // GM→L1
        asc_sync_notify(PIPE_MTE2, PIPE_MTE1, EVENT_ID0);  // MTE2通知MTE1
        asc_sync_wait(PIPE_MTE2, PIPE_MTE1, EVENT_ID0);    // MTE1等待MTE2
        
        // ===== L1→L0A/L0B =====
        asc_copy_l12l0a(a_l0a, a_l1, 0, 1, 1, 0);  // L1→L0A
        asc_copy_l12l0b(b_l0b, b_l1, 0, 1, 1, 0);  // L1→L0B
        asc_sync_notify(PIPE_MTE1, PIPE_CUBE, EVENT_ID0);  // MTE1通知Cube
        asc_sync_wait(PIPE_MTE1, PIPE_CUBE, EVENT_ID0);    // Cube等待MTE1
        
        // ===== 第三步：计算执行 =====
        asc_mmad(c_l0c, a_l0a, b_l0b, BLOCK_M, BLOCK_K, BLOCK_N, 0, false, false, false);  // 矩阵乘
        asc_sync_notify(PIPE_CUBE, PIPE_MTE3, EVENT_ID0);  // Cube通知MTE3
        asc_sync_wait(PIPE_CUBE, PIPE_MTE3, EVENT_ID0);    // MTE3等待Cube
        
        // ===== 第四步：结果搬出 =====
        asc_copy_l0c2gm(c_gm, c_l0c, BLOCK_M * BLOCK_N);  // L0C→GM

        // Cube核完成当前tile写回后，通过核间同步通知Vector核可以开始读取该tile结果。
        asc_sync_block_arrive(PIPE_FIX, 0);
    }

     if ASC_IS_AIV {
        constexpr uint32_t COMPUTE_LEN = BLOCK_M / 2 * BLOCK_N;
        __ubuf__ half src_local[COMPUTE_LEN];
        __ubuf__ half dst_local[COMPUTE_LEN];
        // Vector核需要先等待Cube核完成Fixpipe写回，再开始读取当前tile结果。
        asc_sync_block_wait(PIPE_S, 0);
        // 向量计算逻辑中，cube:vector=1:2，每个Vector核处理BLOCK_M / 2 * BLOCK_N大小的半块结果。
        
        asc_copy_gm2ub(src_local, c_gm + asc_get_block_idx() * COMPUTE_LEN, COMPUTE_LEN);
        asc_leakyrelu(dst_local, src_local, COMPUTE_LEN);
     }
}
```
